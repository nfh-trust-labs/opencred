# ---------- Server ----------
NODE_ENV=production
PORT=3000
LOG_LEVEL=info

# ---------- CORS ----------
# Origin allowed to make API requests (Web UI URL)

# JWT_SECRET=
JWT_ISSUER=opencred
JWT_EXPIRY_SECONDS=3600

# ---------- Batch Processing ----------
MAX_BATCH_SIZE=1000
